cmake_minimum_required(VERSION 3.22)
project(dod3_android_wrapper LANGUAGES C CXX)

set(ROOT_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
set(SDL2_SRC "${ROOT_DIR}/third_party/SDL2")
set(GLSLANG_SRC "${ROOT_DIR}/third_party/glslang")
set(UPSTREAM_DIR "${ROOT_DIR}/upstream")

if(NOT EXISTS "${SDL2_SRC}/CMakeLists.txt")
    message(FATAL_ERROR "SDL2 not found in ${SDL2_SRC}. Run Gradle preBuild/fetchThirdParty first.")
endif()
if(NOT EXISTS "${GLSLANG_SRC}/CMakeLists.txt")
    message(FATAL_ERROR "glslang not found in ${GLSLANG_SRC}. Run Gradle preBuild/fetchThirdParty first.")
endif()
if(NOT EXISTS "${UPSTREAM_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "upstream/ not found. Run bash tools/sync_upstream.sh first.")
endif()

# 1. Build SDL2 (shared libSDL2.so for SDLActivity)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDL2_SRC}" "${CMAKE_CURRENT_BINARY_DIR}/sdl2_build" EXCLUDE_FROM_ALL)

# Expose <SDL2/SDL.h> wrapper headers matching Linux/macOS include style
set(SDL2_SHIM_DIR "${CMAKE_CURRENT_BINARY_DIR}/sdl2_shim")
file(MAKE_DIRECTORY "${SDL2_SHIM_DIR}/SDL2")
file(GLOB SDL2_HDRS "${SDL2_SRC}/include/*.h")
foreach(_h IN LISTS SDL2_HDRS)
    get_filename_component(_hn "${_h}" NAME)
    configure_file("${_h}" "${SDL2_SHIM_DIR}/SDL2/${_hn}" COPYONLY)
endforeach()
include_directories("${SDL2_SHIM_DIR}")

# 2. Build glslang + SPIRV-Tools (static, in-process HLSL -> SPIR-V)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT ON CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(SPIRV_SKIP_TESTS ON CACHE BOOL "" FORCE)
set(SPIRV_SKIP_EXECUTABLES ON CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
add_subdirectory("${GLSLANG_SRC}" "${CMAKE_CURRENT_BINARY_DIR}/glslang_build" EXCLUDE_FROM_ALL)

# Ensure expected CMake aliases exist
if(TARGET glslang AND NOT TARGET glslang::glslang)
    add_library(glslang::glslang ALIAS glslang)
endif()
if(TARGET SPIRV AND NOT TARGET glslang::SPIRV)
    add_library(glslang::SPIRV ALIAS SPIRV)
endif()
if(TARGET glslang-default-resource-limits AND NOT TARGET glslang::glslang-default-resource-limits)
    add_library(glslang::glslang-default-resource-limits ALIAS glslang-default-resource-limits)
endif()

# 3. Build dod3-recomp + ps3recomp_runtime -> libmain.so
add_subdirectory("${UPSTREAM_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dod3_build")
